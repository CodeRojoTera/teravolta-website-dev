**Wire save/cancel handlers for admin quote edits:**

1. **Add onSubmit handler:**
   ```typescript
   const onSaveQuote = async (data: QuoteFormData) => {
     setUpdating(true);
     try {
       const updatePayload: any = {
         client_name: data.clientName,
         client_email: data.clientEmail,
         client_phone: data.clientPhone,
         client_company: data.clientCompany || null,
         address: data.address,
         city: data.city || null,
         state: data.state || null,
         zip_code: data.zipCode || null,
         property_type: data.propertyType,
         property_size: (data as any).propertySize || null,
         operating_hours: (data as any).operatingHours || null,
         booking_date: (data as any).bookingDate || null,
         booking_time: (data as any).bookingTime || null,
          ...(data.service === 'consulting' ? {
            timeline: (data as any).timeline || null,
            budget: (data as any).budget || null,
            project_description: (data as any).projectDescription || null,
          } : {
            timeline: null,
            budget: null,
            project_description: null,
          }),
          ...(data.service === 'advocacy' ? {
            claim_type: (data as any).claimType || null,
            distributor_company: (data as any).distributorCompany || null,
            claim_amount: (data as any).claimAmount || null,
            incident_date: (data as any).incidentDate || null,
            damage_description: (data as any).damageDescription || null
          } : {
            claim_type: null,
            distributor_company: null,
            claim_amount: null,
            incident_date: null,
            damage_description: null
          })
        };

       const { error } = await supabase
         .from('quotes')
         .update(updatePayload)
         .eq('id', quote.id);

       if (error) throw error;
       setQuote((prev: any) => ({ ...prev, ...data }));
       setIsEditingQuote(false);
       showToast('Quote updated', 'success');
     } catch (error) {
       console.error('Error updating quote', error);
       showToast('Error updating quote', 'error');
     } finally {
       setUpdating(false);
     }
   };
   ```

2. **Add cancel handler:**

     reset();
     setIsEditingQuote(false);
   };
