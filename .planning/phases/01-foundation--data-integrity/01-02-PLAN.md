<objective>
Add CASCADE ON DELETE constraints to all relevant foreign keys.

Purpose: Ensure database-level integrity so that when a user or project is deleted, all related entities are automatically cleaned up without requiring application-level cascade logic. This prevents orphaned records (DATA-02, SCHEMA-02, SCHEMA-03).

Output: Single migration that audits existing FKs and adds CASCADE constraints where missing.
</objective>

Run this query against the Supabase database:

```sql
SELECT
  tc.table_name AS child_table,
  kcu.column_name AS fk_column,
  ccu.table_name AS parent_table,
  ccu.column_name AS parent_column,
  rc.delete_rule AS current_delete_behavior
FROM information_schema.table_constraints tc
JOIN information_schema.key_column_usage kcu
  ON tc.constraint_name = kcu.constraint_name
  AND tc.table_schema = kcu.table_schema
JOIN information_schema.referential_constraints rc
  ON tc.constraint_name = rc.constraint_name
  AND tc.table_schema = rc.constraint_schema
JOIN information_schema.constraint_column_usage ccu
  ON rc.unique_constraint_name = ccu.constraint_name
  AND rc.unique_constraint_schema = ccu.constraint_schema
WHERE tc.constraint_type = 'FOREIGN KEY'
  AND tc.table_schema = 'public'
ORDER BY parent_table, child_table;
```

**Template structure (adapt based on audit findings):**

```sql
-- ============================================================================
-- Migration: Add CASCADE ON DELETE to Foreign Keys
-- Purpose: Ensure database-level cascade for user and project deletion
-- Requirements: DATA-02, SCHEMA-02, SCHEMA-03
-- ============================================================================

-- Start transaction
BEGIN;

-- ============================================================================
-- 1. User -> Quotes Cascade
-- ============================================================================

-- Drop existing constraint if exists (use actual constraint name from audit)
ALTER TABLE quotes
  DROP CONSTRAINT IF EXISTS quotes_user_id_fkey;

ALTER TABLE quotes
  ADD CONSTRAINT quotes_user_id_fkey
  FOREIGN KEY (user_id)
  REFERENCES users(uid)
  ON DELETE CASCADE;

-- ============================================================================
-- 2. User -> Active Projects Cascade
-- ============================================================================

ALTER TABLE active_projects
  DROP CONSTRAINT IF EXISTS active_projects_user_id_fkey;

ALTER TABLE active_projects
  ADD CONSTRAINT active_projects_user_id_fkey
  FOREIGN KEY (user_id)
  REFERENCES users(uid)
  ON DELETE CASCADE;

-- ============================================================================
-- 3. Project -> Appointments Cascade (SCHEMA-03)
-- ============================================================================

ALTER TABLE appointments
  DROP CONSTRAINT IF EXISTS appointments_project_id_fkey;

ALTER TABLE appointments
  ADD CONSTRAINT appointments_project_id_fkey
  FOREIGN KEY (project_id)
  REFERENCES active_projects(id)
  ON DELETE CASCADE;

-- ============================================================================
-- 4. User -> Notifications Cascade (if notifications table exists)
-- ============================================================================

DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM information_schema.tables WHERE table_name = 'notifications') THEN
    ALTER TABLE notifications
      DROP CONSTRAINT IF EXISTS notifications_user_id_fkey;

    ALTER TABLE notifications
      ADD CONSTRAINT notifications_user_id_fkey
      FOREIGN KEY (user_id)
      REFERENCES users(uid)
      ON DELETE CASCADE;
  END IF;
END $$;

-- ============================================================================
-- 5. Appointment -> Electrical Boards Cascade (verify/ensure)
-- ============================================================================

-- Check if already CASCADE from 20260114180000_create_electrical_boards.sql
-- If not, add it:
ALTER TABLE electrical_boards
  DROP CONSTRAINT IF EXISTS electrical_boards_appointment_id_fkey;

ALTER TABLE electrical_boards
  ADD CONSTRAINT electrical_boards_appointment_id_fkey
  FOREIGN KEY (appointment_id)
  REFERENCES appointments(id)
  ON DELETE CASCADE;

-- ============================================================================
-- 6. Documents -> Users Cascade (uploaded_by)
-- Note: This allows documents to exist after user deletion (SET NULL might be better)
-- Decision: SET NULL for uploaded_by to preserve audit trail
-- ============================================================================

-- Documents uploaded_by should SET NULL (preserve document, remove uploader reference)
-- This is intentionally NOT CASCADE - documents remain for project history

COMMIT;
```
