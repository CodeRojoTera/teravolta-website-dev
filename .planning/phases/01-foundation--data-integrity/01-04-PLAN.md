import { z } from 'zod';

/**
 * Reasons for user deletion
 */
export const deletionReasonSchema = z.enum([
  'user_request',      // User requested their own deletion
  'admin_action',      // Admin decided to remove user
  'inactivity',        // User inactive for extended period
  'violation',         // User violated terms of service
  'duplicate_account', // Removing duplicate account
  'test_account'       // Removing test data
]);

export type DeletionReason = z.infer<typeof deletionReasonSchema>;

/**
 * Schema for scheduling soft deletion (15-day grace period)
 */
export const scheduleDeletionSchema = z.object({
  userId: z.string().uuid('Invalid user ID format'),
  reason: deletionReasonSchema,
  notifyUser: z.boolean().default(true),
  notifyAdmins: z.boolean().default(true)
});

export type ScheduleDeletionInput = z.infer<typeof scheduleDeletionSchema>;

/**
 * Schema for cancelling scheduled deletion
 */
export const cancelDeletionSchema = z.object({
  userId: z.string().uuid('Invalid user ID format'),
  cancelledBy: z.string().uuid('Invalid admin ID format').optional()
});

export type CancelDeletionInput = z.infer<typeof cancelDeletionSchema>;

/**
 * Schema for immediate hard deletion (admin only)
 */
export const hardDeleteUserSchema = z.object({
  userId: z.string().uuid('Invalid user ID format'),

// app/services/userDeletionService.ts
'use server'

import { createClient } from '@/lib/supabase-admin';
import { addDays } from 'date-fns';
import type {
  DeletionReason,
  ScheduleDeletionInput,
  HardDeleteUserInput
} from '@/lib/validations/deletion-requests';

const GRACE_PERIOD_DAYS = 15; // Per CONTEXT.md

interface DeletionResult {
  success: boolean;
  error?: string;
  deletedEntities?: {
    quotes: number;
    projects: number;
    appointments: number;
    documents: number;
    notifications: number;
  };
  auditLogId?: string;
}

export const UserDeletionService = {

  /**
   * Schedule user for soft deletion with 15-day grace period.
   * User can still log in during grace period with warning banner.
   */
  async scheduleDeletion(input: ScheduleDeletionInput, adminId: string): Promise<DeletionResult> {
    const supabase = createClient();
    const scheduledFor = addDays(new Date(), GRACE_PERIOD_DAYS);

    try {
      // Check if user exists and is not already scheduled for deletion
      const { data: user, error: userError } = await supabase
        .from('users')
        .select('uid, email, full_name, deleted_at, deletion_scheduled_for')
        .eq('uid', input.userId)
        .single();

      if (userError || !user) {
        return { success: false, error: 'User not found' };
      }

      if (user.deleted_at) {
        return { success: false, error: 'User is already deleted' };
      }

      if (user.deletion_scheduled_for) {
        return { success: false, error: 'User deletion is already scheduled' };
      }

      // Check for active projects - put them on hold
      // Use user_id NOT client_id (DATA-01 fix)
      const { data: activeProjects } = await supabase
        .from('active_projects')
        .select('id, status')
        .eq('user_id', input.userId)  // CORRECT: user_id
        .not('status', 'in', '("completed","cancelled")');

      if (activeProjects && activeProjects.length > 0) {
        // Put active projects on hold during grace period
        await supabase
          .from('active_projects')
          .update({
            status: 'on_hold',
            updated_at: new Date().toISOString()
          })
          .eq('user_id', input.userId)  // CORRECT: user_id
          .not('status', 'in', '("completed","cancelled")');
      }

      // Schedule the deletion
      const { error: updateError } = await supabase
        .from('users')
        .update({
          deletion_scheduled_for: scheduledFor.toISOString(),
          deleted_by: adminId
        })
        .eq('uid', input.userId);

      if (updateError) {
        throw updateError;
      }

      // Create audit log entry
      const { data: auditLog, error: auditError } = await supabase
        .from('deletion_audit_log')
        .insert({
          table_name: 'users',
          record_id: input.userId,
          record_data: {
            email: user.email,
            full_name: user.full_name,
            scheduled_for: scheduledFor.toISOString(),
            projects_on_hold: activeProjects?.length || 0
          },
          deleted_by: adminId,
          delete_reason: input.reason,
          deletion_type: 'scheduled'
        })
        .select('id')
        .single();

      if (auditError) {
        console.error('Failed to create audit log:', auditError);
      }

      // TODO: Send notifications (email + dashboard) per CONTEXT.md
      // if (input.notifyUser) { await sendDeletionScheduledEmail(user.email, scheduledFor); }
      // if (input.notifyAdmins) { await createAdminNotification(...); }

      return {
        success: true,
        auditLogId: auditLog?.id,
        deletedEntities: {
          quotes: 0,
          projects: activeProjects?.length || 0, // Projects on hold, not deleted yet
          appointments: 0,
          documents: 0,
          notifications: 0
        }
      };

    } catch (error) {
      console.error('Schedule deletion error:', error);
      return { success: false, error: 'Failed to schedule deletion' };
    }
  },

  /**
   * Cancel a scheduled deletion during grace period.
   * Reactivates any projects that were put on hold.
   */
  async cancelDeletion(userId: string, cancelledBy?: string): Promise<DeletionResult> {
    const supabase = createClient();

    try {
      // Verify user has scheduled deletion
      const { data: user, error: userError } = await supabase
        .from('users')
        .select('uid, deletion_scheduled_for')
        .eq('uid', userId)
        .single();

      if (userError || !user) {
        return { success: false, error: 'User not found' };
      }

      if (!user.deletion_scheduled_for) {
        return { success: false, error: 'No deletion scheduled for this user' };
      }

      // Clear the scheduled deletion
      const { error: updateError } = await supabase
        .from('users')
        .update({
          deletion_scheduled_for: null,
          deleted_by: null
        })
        .eq('uid', userId);

      if (updateError) {
        throw updateError;
      }

      // Reactivate projects that were put on hold
      // Note: This restores to 'in_progress' - might need smarter logic
      // to restore to previous status in production
      await supabase
        .from('active_projects')
        .update({
          status: 'in_progress', // Or could store previous status in metadata
          updated_at: new Date().toISOString()
        })
        .eq('user_id', userId)  // CORRECT: user_id
        .eq('status', 'on_hold');

      // Log the cancellation
      await supabase
        .from('deletion_audit_log')
        .insert({
          table_name: 'users',
          record_id: userId,
          record_data: { action: 'cancellation', cancelled_by: cancelledBy },
          deleted_by: cancelledBy || null,
          delete_reason: 'user_request',
          deletion_type: 'soft' // Using 'soft' to indicate it was a soft delete that was cancelled
        });

      return { success: true };

    } catch (error) {
      console.error('Cancel deletion error:', error);
      return { success: false, error: 'Failed to cancel deletion' };
    }
  },

  /**
   * Immediately and permanently hard delete a user.
   * Admin only - bypasses grace period.
   * Requires confirmation and reason.
   */
  async hardDelete(input: HardDeleteUserInput, adminId: string): Promise<DeletionResult> {
    const supabase = createClient();

    try {
      // Get user data for audit log BEFORE deletion
      const { data: user, error: userError } = await supabase
        .from('users')
        .select('*')
        .eq('uid', input.userId)
        .single();

      if (userError || !user) {
        return { success: false, error: 'User not found' };
      }

      // Count related entities for audit (before CASCADE deletes them)
      const [quotesResult, projectsResult, appointmentsResult, docsResult, notifsResult] = await Promise.all([
        supabase.from('quotes').select('id', { count: 'exact' }).eq('user_id', input.userId),
        supabase.from('active_projects').select('id', { count: 'exact' }).eq('user_id', input.userId),
        supabase.from('appointments').select('id', { count: 'exact' }).eq('client_user_id', input.userId),
        supabase.from('documents').select('id', { count: 'exact' }).eq('uploaded_by', input.userId),
        supabase.from('notifications').select('id', { count: 'exact' }).eq('user_id', input.userId)
      ]);

      const deletedEntities = {
        quotes: quotesResult.count || 0,
        projects: projectsResult.count || 0,
        appointments: appointmentsResult.count || 0,
        documents: docsResult.count || 0,
        notifications: notifsResult.count || 0
      };

      // Create comprehensive audit log BEFORE deletion
      const { data: auditLog, error: auditError } = await supabase
        .from('deletion_audit_log')
        .insert({
          table_name: 'users',
          record_id: input.userId,
          record_data: {
            ...user,
            related_entities: deletedEntities
          },
          deleted_by: adminId,
          delete_reason: input.reason,
          deletion_type: 'hard',
          related_deletions: [
            ...Array(deletedEntities.quotes).fill({ table: 'quotes' }),
            ...Array(deletedEntities.projects).fill({ table: 'active_projects' }),
            ...Array(deletedEntities.appointments).fill({ table: 'appointments' }),
            ...Array(deletedEntities.documents).fill({ table: 'documents' }),
            ...Array(deletedEntities.notifications).fill({ table: 'notifications' })
          ]
        })
        .select('id')
        .single();

      if (auditError) {
        console.error('Failed to create audit log:', auditError);
        // Continue with deletion - audit failure shouldn't block
      }

      // Delete from Supabase Auth first (this is the actual user account)
      const { error: authDeleteError } = await supabase.auth.admin.deleteUser(input.userId);

      if (authDeleteError) {
        // Log but potentially continue - auth user might not exist
        console.error('Auth delete error:', authDeleteError);
      }

      // Delete from users table - CASCADE will handle related entities
      // Use user_id NOT client_id (DATA-01 fix verified)
      const { error: deleteError } = await supabase
        .from('users')
        .delete()
        .eq('uid', input.userId);  // CORRECT field name

      if (deleteError) {
        throw deleteError;
      }

      return {
        success: true,
        auditLogId: auditLog?.id,
        deletedEntities
      };

    } catch (error) {
      console.error('Hard delete error:', error);
      return { success: false, error: 'Failed to delete user' };
    }
  },

  /**
   * Execute soft delete (called by scheduled job after grace period).
   * Converts scheduled deletion to actual soft delete.
   */
  async executeSoftDelete(userId: string): Promise<DeletionResult> {
    const supabase = createClient();

    try {
      const { data: user } = await supabase
        .from('users')
        .select('uid, deletion_scheduled_for, deleted_by')
        .eq('uid', userId)
        .single();

      if (!user || !user.deletion_scheduled_for) {
        return { success: false, error: 'User not scheduled for deletion' };
      }

      // Check if grace period has passed
      if (new Date(user.deletion_scheduled_for) > new Date()) {
        return { success: false, error: 'Grace period has not ended' };
      }

      // Perform soft delete
      const { error: updateError } = await supabase
        .from('users')
        .update({
          deleted_at: new Date().toISOString()
        })
        .eq('uid', userId);

      if (updateError) {
        throw updateError;
      }

      // Soft delete related projects (DATA-04)
      await supabase
        .from('active_projects')
        .update({
          status: 'cancelled',
          updated_at: new Date().toISOString()
        })
        .eq('user_id', userId);  // CORRECT: user_id

      // Update audit log
      await supabase
        .from('deletion_audit_log')
        .insert({
          table_name: 'users',
          record_id: userId,
          record_data: { action: 'soft_delete_executed' },
          deleted_by: user.deleted_by,
          delete_reason: 'user_request', // Was scheduled, so originally user request
          deletion_type: 'soft'
        });

      return { success: true };

    } catch (error) {
      console.error('Execute soft delete error:', error);
      return { success: false, error: 'Failed to execute soft delete' };
    }
  },

  /**
   * Reactivate a soft-deleted user (admin only).
   * Per CONTEXT.md: Users deleted >7 days require password reset.
   */
  async reactivateUser(userId: string, adminId: string): Promise<DeletionResult> {
    const supabase = createClient();

          deleted_at: null,

        })
        .eq('uid', userId);

      if (updateError) {
        throw updateError;
      }

      // Log reactivation
      await supabase
        .from('deletion_audit_log')
        .insert({
          table_name: 'users',
          record_id: userId,
          record_data: {
            action: 'reactivation',
            reactivated_by: adminId,
            requires_password_reset: requiresPasswordReset
          },
          deleted_by: adminId,
          delete_reason: 'admin_action',
          deletion_type: 'soft'
        });

      // If requires password reset, trigger it

      }

  }
};

// app/api/admin/delete-user/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase-admin';
import { UserDeletionService } from '@/app/services/userDeletionService';
import { hardDeleteUserSchema } from '@/lib/validations/deletion-requests';

export async function POST(request: NextRequest) {
  try {
    const supabase = createClient();

    // Verify admin authentication
    const { data: { user }, error: authError } = await supabase.auth.getUser();

    if (authError || !user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Verify user is admin
    const { data: adminUser } = await supabase
      .from('users')
      .select('role')
      .eq('uid', user.id)
      .single();

    if (!adminUser || !['admin', 'super_admin'].includes(adminUser.role)) {
      return NextResponse.json({ error: 'Admin access required' }, { status: 403 });
    }

    // Parse and validate request body
    const body = await request.json();
    const parsed = hardDeleteUserSchema.safeParse(body);

    if (!parsed.success) {
      return NextResponse.json({
        error: 'Validation failed',
        details: parsed.error.flatten()
      }, { status: 400 });
    }

    // Prevent self-deletion
    if (parsed.data.userId === user.id) {
      return NextResponse.json({
        error: 'Cannot delete your own account'
      }, { status: 400 });
    }

    // Perform hard delete
    const result = await UserDeletionService.hardDelete(parsed.data, user.id);

    if (!result.success) {
      return NextResponse.json({ error: result.error }, { status: 400 });
    }

    return NextResponse.json({
      success: true,
      message: 'User permanently deleted',
      deletedEntities: result.deletedEntities,
      auditLogId: result.auditLogId
    });

  } catch (error) {
    console.error('Delete user API error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}

import { createClient } from '@/lib/supabase-admin';
import { UserDeletionService } from '@/app/services/userDeletionService';
import { cancelDeletionSchema } from '@/lib/validations/deletion-requests';

export async function POST(request: NextRequest) {
  try {
    const supabase = createClient();

    // This can be called by admin OR by the user themselves
    const { data: { user }, error: authError } = await supabase.auth.getUser();

    if (authError || !user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const body = await request.json();
    const parsed = cancelDeletionSchema.safeParse(body);

    if (!parsed.success) {
      return NextResponse.json({
        error: 'Validation failed',
        details: parsed.error.flatten()
      }, { status: 400 });
    }

    // If user is cancelling their own deletion, allow it
    // If admin is cancelling for another user, verify admin role
    if (parsed.data.userId !== user.id) {
      const { data: adminUser } = await supabase
        .from('users')
        .select('role')
        .eq('uid', user.id)
        .single();

      if (!adminUser || !['admin', 'super_admin'].includes(adminUser.role)) {
        return NextResponse.json({ error: 'Cannot cancel deletion for other users' }, { status: 403 });
      }
    }

    // Cancel the scheduled deletion
    const result = await UserDeletionService.cancelDeletion(parsed.data.userId, user.id);

    if (!result.success) {
      return NextResponse.json({ error: result.error }, { status: 400 });
    }

    return NextResponse.json({
      success: true,
      message: 'User deletion cancelled successfully'
    });

  } catch (error) {
    console.error('Cancel deletion API error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}
