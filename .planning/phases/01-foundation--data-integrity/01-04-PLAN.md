// lib/services/deletion-audit.ts
import { createClient } from '@/lib/supabase-admin';

export type DeletionType = 'soft' | 'hard' | 'cascade' | 'scheduled';
export type DeletionReason = 'user_request' | 'admin_action' | 'inactivity' | 'violation' | 'system' | 'gdpr';

export interface DeletionLogEntry {
  tableName: string;
  recordId: string;
  recordData: Record<string, unknown>;
  deletedBy: string | null;
  deletionType: DeletionType;
  deletionReason: DeletionReason;
  parentDeletionId?: string;
  notes?: string;
}

/**
 * Log a deletion to the audit log.
 * Call this BEFORE actually deleting the record to capture its full state.
 */
export async function logDeletion(entry: DeletionLogEntry): Promise<string | null> {
  const supabase = createClient();

  const { data, error } = await supabase
    .from('deletion_audit_log')
    .insert({
      table_name: entry.tableName,
      record_id: entry.recordId,
      record_data: entry.recordData,
      deleted_by: entry.deletedBy,
      deletion_type: entry.deletionType,
      deletion_reason: entry.deletionReason,
      parent_deletion_id: entry.parentDeletionId,
      notes: entry.notes,
    })
    .select('id')
    .single();

  if (error) {
    console.error('Failed to log deletion:', error);
    return null;
  }

  return data?.id || null;
}

/**
 * Log cascade deletions (child records deleted due to parent deletion).
 * Updates the parent deletion record with related deletion IDs.
 */
export async function logCascadeDeletions(
  parentDeletionId: string,
  cascadeEntries: Omit<DeletionLogEntry, 'parentDeletionId' | 'deletionType'>[]
): Promise<string[]> {
  const childIds: string[] = [];

  for (const entry of cascadeEntries) {
    const id = await logDeletion({
      ...entry,
      deletionType: 'cascade',
      parentDeletionId,
    });
    if (id) childIds.push(id);
  }

  // Update parent with related deletions
  if (childIds.length > 0) {
    const supabase = createClient();
    await supabase
      .from('deletion_audit_log')
      .update({ related_deletions: childIds })
      .eq('id', parentDeletionId);
  }

  return childIds;
}

/**
 * Fetch user's complete data for audit log before deletion.
 * Returns all related records that will be affected.
 */
export async function fetchUserDataForAudit(userId: string): Promise<{
  user: Record<string, unknown> | null;
  projects: Record<string, unknown>[];
  appointments: Record<string, unknown>[];
  documents: Record<string, unknown>[];
  invoices: Record<string, unknown>[];
  notifications: Record<string, unknown>[];
}> {
  const supabase = createClient();

  const [userResult, projectsResult, appointmentsResult, documentsResult, invoicesResult, notificationsResult] = await Promise.all([
    supabase.from('users').select('*').eq('id', userId).single(),
    supabase.from('active_projects').select('*').eq('user_id', userId),
    supabase.from('appointments').select('*').in('project_id',
      supabase.from('active_projects').select('id').eq('user_id', userId)
    ),
    supabase.from('documents').select('*').or(`user_id.eq.${userId},uploaded_by.eq.${userId}`),
    supabase.from('invoices').select('*').eq('user_id', userId),
    supabase.from('notifications').select('*').eq('user_id', userId),
  ]);

  return {
    user: userResult.data,
    projects: projectsResult.data || [],
    appointments: appointmentsResult.data || [],
    documents: documentsResult.data || [],
    invoices: invoicesResult.data || [],
    notifications: notificationsResult.data || [],
  };
}

// lib/services/user-deletion.ts
import { createClient } from '@/lib/supabase-admin';
import { addDays } from 'date-fns';
import { logDeletion, logCascadeDeletions, fetchUserDataForAudit, DeletionReason } from './deletion-audit';

const GRACE_PERIOD_DAYS = 15;

export interface DeletionResult {
  success: boolean;
  error?: string;
  type?: 'scheduled' | 'soft' | 'hard';
  scheduledFor?: string;
  auditLogId?: string;
}

export interface UserDeletionOptions {
  userId: string;
  deletedBy: string;
  reason: DeletionReason;
  notes?: string;
  immediate?: boolean; // Skip grace period (admin override)
  forceDelete?: boolean; // Delete even if user has active projects
}

/**
 * Check if user has active projects that would block deletion.
 */
export async function checkUserHasActiveProjects(userId: string): Promise<{
  hasActive: boolean;
  count: number;
  projects: { id: string; status: string; client_name: string }[];
}> {
  const supabase = createClient();

  const { data, error } = await supabase
    .from('active_projects')
    .select('id, status, client_name')
    .eq('user_id', userId)
    .not('status', 'in', '("completed","cancelled")');

  if (error) {
    console.error('Error checking active projects:', error);
    return { hasActive: false, count: 0, projects: [] };
  }

  return {
    hasActive: (data?.length || 0) > 0,
    count: data?.length || 0,
    projects: data || [],
  };
}

/**
 * Schedule user deletion with 15-day grace period.
 * User can cancel during grace period. Projects are put on hold.
 */
export async function scheduleUserDeletion(options: UserDeletionOptions): Promise<DeletionResult> {
  const { userId, deletedBy, reason, notes, forceDelete } = options;
  const supabase = createClient();

  // Check for active projects
  const { hasActive, count, projects } = await checkUserHasActiveProjects(userId);

  if (hasActive && !forceDelete) {
    return {
      success: false,
      error: `User has ${count} active project(s). Use forceDelete to proceed or cancel/complete projects first.`,
    };
  }

  const scheduledFor = addDays(new Date(), GRACE_PERIOD_DAYS);

  // Update user with scheduled deletion
  const { error: updateError } = await supabase
    .from('users')
    .update({
      deletion_scheduled_for: scheduledFor.toISOString(),
      deleted_by: deletedBy,
    })
    .eq('id', userId);

  if (updateError) {
    return { success: false, error: updateError.message };
  }

  // Put active projects on hold
  if (hasActive) {
    await supabase
      .from('active_projects')
      .update({ status: 'on_hold' })
      .eq('user_id', userId)
      .not('status', 'in', '("completed","cancelled")');
  }

  // Log the scheduled deletion
  const auditId = await logDeletion({
    tableName: 'users',
    recordId: userId,
    recordData: { scheduled_for: scheduledFor.toISOString(), projects_affected: projects },
    deletedBy,
    deletionType: 'scheduled',
    deletionReason: reason,
    notes: notes || `Deletion scheduled for ${scheduledFor.toISOString()}. ${count} project(s) put on hold.`,
  });

  return {
    success: true,
    type: 'scheduled',
    scheduledFor: scheduledFor.toISOString(),
    auditLogId: auditId || undefined,
  };
}

/**
 * Cancel a scheduled deletion (restore user to active state).
 */
export async function cancelScheduledDeletion(userId: string, cancelledBy: string): Promise<DeletionResult> {
  const supabase = createClient();

  // Clear scheduled deletion
  const { error } = await supabase
    .from('users')
    .update({
      deletion_scheduled_for: null,
      deleted_by: null,
    })
    .eq('id', userId)
    .not('deleted_at', 'is', null); // Only if not already deleted

  if (error) {
    return { success: false, error: error.message };
  }

  // Log cancellation
  await logDeletion({
    tableName: 'users',
    recordId: userId,
    recordData: { action: 'cancellation', cancelled_by: cancelledBy },
    deletedBy: cancelledBy,
    deletionType: 'soft',
    deletionReason: 'user_request',
    notes: 'Scheduled deletion cancelled',
  });

  return { success: true };
}

/**
 * Execute soft delete (set deleted_at, preserve record).
 * Used when grace period expires or admin chooses soft delete.
 */
export async function executeSoftDelete(options: UserDeletionOptions): Promise<DeletionResult> {
  const { userId, deletedBy, reason, notes } = options;
  const supabase = createClient();

  // Fetch full user data for audit
  const userData = await fetchUserDataForAudit(userId);

  if (!userData.user) {
    return { success: false, error: 'User not found' };
  }

  // Log the deletion with full snapshot
  const auditId = await logDeletion({
    tableName: 'users',
    recordId: userId,
    recordData: userData.user,
    deletedBy,
    deletionType: 'soft',
    deletionReason: reason,
    notes,
  });

  // Execute soft delete
  const { error } = await supabase
    .from('users')
    .update({
      deleted_at: new Date().toISOString(),
      deleted_by: deletedBy,
      deletion_scheduled_for: null, // Clear if was scheduled
    })
    .eq('id', userId);

  if (error) {
    return { success: false, error: error.message };
  }

  return {
    success: true,
    type: 'soft',
    auditLogId: auditId || undefined,
  };
}

/**
 * Execute immediate hard delete (permanently remove user and all related data).
 * Requires admin privilege and reason. CASCADE constraints handle related records.
 */
export async function executeHardDelete(options: UserDeletionOptions): Promise<DeletionResult> {
  const { userId, deletedBy, reason, notes, forceDelete } = options;
  const supabase = createClient();

  // Check for active projects
  const { hasActive, count } = await checkUserHasActiveProjects(userId);

  if (hasActive && !forceDelete) {
    return {
      success: false,
      error: `User has ${count} active project(s). Set forceDelete to true to proceed.`,
    };
  }

  // Fetch full user data for audit BEFORE deletion
  const userData = await fetchUserDataForAudit(userId);

  if (!userData.user) {
    return { success: false, error: 'User not found' };
  }

  // Log the main deletion
  const auditId = await logDeletion({
    tableName: 'users',
    recordId: userId,
    recordData: userData.user,
    deletedBy,
    deletionType: 'hard',
    deletionReason: reason,
    notes: notes || 'Immediate hard delete by admin',
  });

  // Log cascade deletions for audit trail
  if (auditId) {
    const cascadeEntries = [
      ...userData.projects.map(p => ({
        tableName: 'active_projects',
        recordId: p.id as string,
        recordData: p,
        deletedBy,
        deletionReason: reason,
        notes: 'Cascade deleted with user',
      })),
      ...userData.appointments.map(a => ({
        tableName: 'appointments',
        recordId: a.id as string,
        recordData: a,
        deletedBy,
        deletionReason: reason,
        notes: 'Cascade deleted with user/project',
      })),
      ...userData.invoices.map(i => ({
        tableName: 'invoices',
        recordId: i.id as string,
        recordData: i,
        deletedBy,
        deletionReason: reason,
        notes: 'Cascade deleted with user',
      })),
    ];

    await logCascadeDeletions(auditId, cascadeEntries);
  }

  // Delete from auth.users (this cascades to public.users due to FK)
  // Note: Using admin client for auth operations
  const { error: authError } = await supabase.auth.admin.deleteUser(userId);

  if (authError) {
    // Try deleting from public.users directly if auth deletion fails
    const { error: publicError } = await supabase
      .from('users')
      .delete()
      .eq('id', userId);

    if (publicError) {
      return { success: false, error: `Auth: ${authError.message}, Public: ${publicError.message}` };
    }
  }

  return {
    success: true,
    type: 'hard',
    auditLogId: auditId || undefined,
  };
}

// app/api/admin/users/[id]/delete/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase-admin';
import {
  scheduleUserDeletion,
  executeSoftDelete,
  executeHardDelete,
  cancelScheduledDeletion,
  checkUserHasActiveProjects,
} from '@/lib/services/user-deletion';
import { DeletionReason } from '@/lib/services/deletion-audit';

interface DeleteUserRequest {
  action: 'schedule' | 'soft' | 'hard' | 'cancel';
  reason?: DeletionReason;
  notes?: string;
  forceDelete?: boolean;
}

export async function DELETE(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const supabase = createClient();
    const userId = params.id;

    // Get current user from session
    const { data: { user: currentUser }, error: authError } = await supabase.auth.getUser();

    if (authError || !currentUser) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Check if current user is admin
    const { data: adminUser, error: adminError } = await supabase
      .from('users')
      .select('role')
      .eq('id', currentUser.id)
      .single();

    if (adminError || !adminUser || !['admin', 'super_admin'].includes(adminUser.role)) {
      return NextResponse.json({ error: 'Forbidden: Admin access required' }, { status: 403 });
    }

    // Parse request body
    const body: DeleteUserRequest = await request.json();
    const { action, reason = 'admin_action', notes, forceDelete } = body;

    // Validate reason
    const validReasons: DeletionReason[] = ['user_request', 'admin_action', 'inactivity', 'violation', 'system', 'gdpr'];
    if (!validReasons.includes(reason)) {
      return NextResponse.json({ error: 'Invalid deletion reason' }, { status: 400 });
    }

    // Check if target user exists
    const { data: targetUser, error: targetError } = await supabase
      .from('users')
      .select('id, email, full_name, role')
      .eq('id', userId)
      .single();

    if (targetError || !targetUser) {
      return NextResponse.json({ error: 'User not found' }, { status: 404 });
    }

    // Prevent deleting super_admin unless you are super_admin
    if (targetUser.role === 'super_admin' && adminUser.role !== 'super_admin') {
      return NextResponse.json({ error: 'Cannot delete super_admin' }, { status: 403 });
    }

    // Execute requested action
    let result;

    switch (action) {
      case 'schedule':
        result = await scheduleUserDeletion({
          userId,
          deletedBy: currentUser.id,
          reason,
          notes,
          forceDelete,
        });
        break;

      case 'soft':
        result = await executeSoftDelete({
          userId,
          deletedBy: currentUser.id,
          reason,
          notes,
        });
        break;

      case 'hard':
        // Hard delete requires explicit confirmation
        if (!notes) {
          return NextResponse.json({
            error: 'Hard delete requires notes/reason for audit trail'
          }, { status: 400 });
        }
        result = await executeHardDelete({
          userId,
          deletedBy: currentUser.id,
          reason,
          notes,
          forceDelete,
          immediate: true,
        });
        break;

      case 'cancel':
        result = await cancelScheduledDeletion(userId, currentUser.id);
        break;

      default:
        return NextResponse.json({ error: 'Invalid action' }, { status: 400 });
    }

    if (!result.success) {
      return NextResponse.json({ error: result.error }, { status: 400 });
    }

    return NextResponse.json({
      success: true,
      action,
      ...result,
    });

  } catch (error) {
    console.error('User deletion error:', error);
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    );
  }
}

// GET endpoint to check deletion status and active projects
export async function GET(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const supabase = createClient();
    const userId = params.id;

    // Check authentication
    const { data: { user: currentUser }, error: authError } = await supabase.auth.getUser();

    if (authError || !currentUser) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Get user deletion status
    const { data: user, error: userError } = await supabase
      .from('users')
      .select('id, email, full_name, deleted_at, deletion_scheduled_for, deleted_by')
      .eq('id', userId)
      .single();

    if (userError || !user) {
      return NextResponse.json({ error: 'User not found' }, { status: 404 });
    }

    // Check for active projects
    const activeProjects = await checkUserHasActiveProjects(userId);

    return NextResponse.json({
      user: {
        id: user.id,
        email: user.email,
        fullName: user.full_name,
        deletedAt: user.deleted_at,
        deletionScheduledFor: user.deletion_scheduled_for,
      },
      activeProjects,
    });

  } catch (error) {
    console.error('Get user deletion status error:', error);
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    );
  }
}
