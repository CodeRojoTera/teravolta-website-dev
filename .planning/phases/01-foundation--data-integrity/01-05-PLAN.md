```typescript
/**
 * Update project status with state machine validation.
 * @param projectId - The project ID
 * @param newStatus - The desired new status
 * @param isAdmin - Whether the user is an admin (allows override)
 * @param overrideReason - Required if admin is overriding invalid transition
 * @returns Result with success flag and any error message
 */
async updateStatus(
  projectId: string,
  newStatus: ProjectStatus,
  isAdmin: boolean = false,
  overrideReason?: string
): Promise<{ success: boolean; error?: string; previousStatus?: string }> {
  const supabase = getBrowserClient();

  try {
    // Get current project
    const { data: project, error: fetchError } = await supabase
      .from('active_projects')
      .select('status, service')
      .eq('id', projectId)
      .single();

    if (fetchError || !project) {
      return { success: false, error: 'Project not found' };
    }

    const currentStatus = project.status as ProjectStatus;
    const service = project.service as ServiceType;

    // Validate transition
    const validationResult = canTransition(currentStatus, newStatus, service, isAdmin);

    if (!validationResult.valid) {
      return {
        success: false,
        error: validationResult.reason || 'Invalid status transition',
        previousStatus: currentStatus
      };
    }

    // Perform the update
    const { error: updateError } = await supabase
      .from('active_projects')
      .update({
        status: newStatus,
        updated_at: new Date().toISOString()
      })
      .eq('id', projectId);

    if (updateError) {
      throw updateError;
    }

    // If this was an admin override, log it
    if (validationResult.reason === 'admin_override') {
      // Log to audit system
      console.warn(
        `Admin override: Project ${projectId} transitioned from ${currentStatus} to ${newStatus}. Reason: ${overrideReason || 'Not provided'}`
      );
      // TODO: Add to proper audit log table when notification system is ready
    }

    return { success: true, previousStatus: currentStatus };

  } catch (error) {
    console.error('Update status error:', error);
    return { success: false, error: 'Failed to update project status' };
  }
}
```

```typescript
// app/api/update-project-status/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase-admin';
import { z } from 'zod';
import {
  canTransition,
  getValidTransitions,
  getStatusLabel
} from '@/lib/state-machines/project-states';
import type { ProjectStatus, ServiceType } from '@/lib/state-machines/types';

// Validation schema
const updateStatusSchema = z.object({
  projectId: z.string().uuid('Invalid project ID'),
  newStatus: z.string().min(1, 'Status is required'),
  forceOverride: z.boolean().optional().default(false),
  overrideReason: z.string().optional()
});

export async function POST(request: NextRequest) {
  try {
    const supabase = createClient();

    // Get current user
    const { data: { user }, error: authError } = await supabase.auth.getUser();

    if (authError || !user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Check if user is admin
    const { data: userData } = await supabase
      .from('users')
      .select('role')
      .eq('uid', user.id)
      .single();

    const isAdmin = userData?.role === 'admin' || userData?.role === 'super_admin';

    // Parse request
    const body = await request.json();
    const parsed = updateStatusSchema.safeParse(body);

    if (!parsed.success) {
      return NextResponse.json({
        error: 'Validation failed',
        details: parsed.error.flatten()
      }, { status: 400 });
    }

    const { projectId, newStatus, forceOverride, overrideReason } = parsed.data;

    // Get current project
    const { data: project, error: projectError } = await supabase
      .from('active_projects')
      .select('id, status, service, user_id, project_name')
      .eq('id', projectId)
      .single();

    if (projectError || !project) {
      return NextResponse.json({ error: 'Project not found' }, { status: 404 });
    }

    // Authorization check: user must own project or be admin
    if (project.user_id !== user.id && !isAdmin) {
      return NextResponse.json({ error: 'Not authorized to update this project' }, { status: 403 });
    }

    const currentStatus = project.status as ProjectStatus;
    const service = project.service as ServiceType;

    // Validate the transition
    const allowOverride = isAdmin && forceOverride;
    const validationResult = canTransition(
      currentStatus,
      newStatus as ProjectStatus,
      service,
      allowOverride
    );

    if (!validationResult.valid) {
      // Get valid options to include in error response
      const validOptions = getValidTransitions(currentStatus, service);

      return NextResponse.json({
        error: 'Invalid status transition',
        details: validationResult.reason,
        currentStatus,
        requestedStatus: newStatus,
        validOptions: validOptions.map(s => ({
          status: s,
          label: getStatusLabel(s, 'en')
        }))
      }, { status: 400 });
    }

    // Perform the update
    const { error: updateError } = await supabase
      .from('active_projects')
      .update({
        status: newStatus,
        updated_at: new Date().toISOString()
      })
      .eq('id', projectId);

    if (updateError) {
      throw updateError;
    }

    // Log admin override if applicable
    if (validationResult.reason === 'admin_override') {
      // Insert into audit log
      await supabase.from('deletion_audit_log').insert({
        table_name: 'active_projects',
        record_id: projectId,
        record_data: {
          action: 'status_override',
          previous_status: currentStatus,
          new_status: newStatus,
          reason: overrideReason || 'No reason provided'
        },
        deleted_by: user.id,
        delete_reason: 'admin_action',
        deletion_type: 'soft' // Reusing audit log, not a deletion
      });

      console.warn(
        `[ADMIN OVERRIDE] User ${user.id} changed project ${projectId} ` +
        `from ${currentStatus} to ${newStatus}. Reason: ${overrideReason || 'Not provided'}`
      );
    }

    return NextResponse.json({
      success: true,
      projectId,
      previousStatus: currentStatus,
      newStatus,
      wasOverride: validationResult.reason === 'admin_override'
    });

  } catch (error) {
    console.error('Update project status error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}

// GET endpoint to retrieve valid transitions for a project
export async function GET(request: NextRequest) {
  try {
    const supabase = createClient();

    // Get current user
    const { data: { user }, error: authError } = await supabase.auth.getUser();

    if (authError || !user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Get project ID from query params
    const { searchParams } = new URL(request.url);
    const projectId = searchParams.get('projectId');

    if (!projectId) {
      return NextResponse.json({ error: 'projectId is required' }, { status: 400 });
    }

    // Get project
    const { data: project, error: projectError } = await supabase
      .from('active_projects')
      .select('id, status, service')
      .eq('id', projectId)
      .single();

    if (projectError || !project) {
      return NextResponse.json({ error: 'Project not found' }, { status: 404 });
    }

    const currentStatus = project.status as ProjectStatus;
    const service = project.service as ServiceType;

    // Get valid transitions
    const validTransitions = getValidTransitions(currentStatus, service);

    return NextResponse.json({
      currentStatus,
      service,
      validTransitions: validTransitions.map(status => ({
        status,
        label: {
          en: getStatusLabel(status, 'en'),
          es: getStatusLabel(status, 'es')
        }
      }))
    });

  } catch (error) {
    console.error('Get valid transitions error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}
```

```typescript
// components/admin/StatusSelector.tsx
'use client'

import { useState, useEffect } from 'react';
import { useLanguage } from '@/components/LanguageProvider';
import {
  getValidTransitions,
  getStatusLabel,
  getStatusColor
} from '@/lib/state-machines/project-states';
import type { ProjectStatus, ServiceType } from '@/lib/state-machines/types';

interface StatusSelectorProps {
  currentStatus: ProjectStatus;
  service: ServiceType;
  projectId: string;
  onStatusChange: (newStatus: ProjectStatus) => Promise<void>;
  isAdmin?: boolean;
  disabled?: boolean;
  showCurrentStatus?: boolean;
}

export function StatusSelector({
  currentStatus,
  service,
  projectId,
  onStatusChange,
  isAdmin = false,
  disabled = false,
  showCurrentStatus = true
}: StatusSelectorProps) {
  const { language } = useLanguage();
  const [selectedStatus, setSelectedStatus] = useState<ProjectStatus | ''>('');
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [showOverrideDialog, setShowOverrideDialog] = useState(false);
  const [overrideReason, setOverrideReason] = useState('');

  // Get valid transitions
  const validTransitions = getValidTransitions(currentStatus, service);
  const hasValidTransitions = validTransitions.length > 0;

  // Handle status selection
  const handleStatusSelect = async (newStatus: ProjectStatus) => {
    if (newStatus === currentStatus) return;

    setError(null);
    setIsLoading(true);

    try {
      await onStatusChange(newStatus);
      setSelectedStatus('');
    } catch (err: any) {
      setError(err.message || 'Failed to update status');
    } finally {
      setIsLoading(false);
    }
  };

  // Handle admin override
  const handleOverride = async (newStatus: ProjectStatus) => {
    if (!overrideReason.trim()) {
      setError('Please provide a reason for the override');
      return;
    }

    setIsLoading(true);
    setError(null);

    try {
      const response = await fetch('/api/update-project-status', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          projectId,
          newStatus,
          forceOverride: true,
          overrideReason
        })
      });

      if (!response.ok) {
        const data = await response.json();
        throw new Error(data.error || 'Failed to update status');
      }

      setShowOverrideDialog(false);
      setOverrideReason('');
      setSelectedStatus('');

      // Reload to reflect change (or call parent refresh)
      window.location.reload();

    } catch (err: any) {
      setError(err.message || 'Failed to override status');
    } finally {
      setIsLoading(false);
    }
  };

  const currentStatusLabel = getStatusLabel(currentStatus, language as 'en' | 'es');
  const currentStatusColor = getStatusColor(currentStatus);

  if (!hasValidTransitions && !isAdmin) {
    // Terminal state, show badge only
    return (
      <div className="flex items-center gap-2">
        <span className={`px-3 py-1 rounded-full text-sm font-medium ${currentStatusColor}`}>
          {currentStatusLabel}
        </span>
        <span className="text-xs text-gray-500">
          {language === 'es' ? '(Estado final)' : '(Final status)'}
        </span>
      </div>
    );
  }

  return (
    <div className="space-y-2">
      {/* Current status display */}
      {showCurrentStatus && (
        <div className="flex items-center gap-2 mb-3">
          <span className="text-sm text-gray-600">
            {language === 'es' ? 'Estado actual:' : 'Current status:'}
          </span>
          <span className={`px-3 py-1 rounded-full text-sm font-medium ${currentStatusColor}`}>
            {currentStatusLabel}
          </span>
        </div>
      )}

      {/* Status selector dropdown */}
      <div className="flex items-center gap-2">
        <select
          value={selectedStatus}
          onChange={(e) => setSelectedStatus(e.target.value as ProjectStatus)}
          disabled={disabled || isLoading}
          className="block w-full rounded-md border-gray-300 shadow-sm focus:border-blue-500 focus:ring-blue-500 sm:text-sm disabled:bg-gray-100"
        >
          <option value="">
            {language === 'es' ? '-- Cambiar estado --' : '-- Change status --'}
          </option>
          {validTransitions.map((status) => (
            <option key={status} value={status}>
              {getStatusLabel(status, language as 'en' | 'es')}
            </option>
          ))}
        </select>

        <button
          onClick={() => selectedStatus && handleStatusSelect(selectedStatus)}
          disabled={!selectedStatus || isLoading || disabled}
          className="px-4 py-2 bg-blue-600 text-white rounded-md hover:bg-blue-700 disabled:bg-gray-300 disabled:cursor-not-allowed transition-colors"
        >
          {isLoading ? (
            <span className="flex items-center gap-2">
              <svg className="animate-spin h-4 w-4" viewBox="0 0 24 24">
                <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" fill="none" />
                <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z" />
              </svg>
              {language === 'es' ? 'Actualizando...' : 'Updating...'}
            </span>
          ) : (
            language === 'es' ? 'Actualizar' : 'Update'
          )}
        </button>
      </div>

      {/* Admin override button (only if admin and terminal state) */}
      {isAdmin && !hasValidTransitions && (
        <button
          onClick={() => setShowOverrideDialog(true)}
          className="text-sm text-amber-600 hover:text-amber-700 underline"
        >
          {language === 'es' ? 'Forzar cambio (Admin)' : 'Force change (Admin)'}
        </button>
      )}

      {/* Error display */}
      {error && (
        <div className="text-sm text-red-600 bg-red-50 px-3 py-2 rounded-md">
          {error}
        </div>
      )}

      {/* Admin override dialog */}
      {showOverrideDialog && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
          <div className="bg-white rounded-lg p-6 max-w-md w-full mx-4 shadow-xl">
            <h3 className="text-lg font-semibold mb-4">
              {language === 'es' ? 'Forzar Cambio de Estado' : 'Force Status Change'}
            </h3>
            <p className="text-sm text-gray-600 mb-4">
              {language === 'es'
                ? 'Esta accion omite las reglas normales de transicion. Por favor proporcione una razon.'
                : 'This action bypasses normal transition rules. Please provide a reason.'}
            </p>

            <div className="mb-4">
              <label className="block text-sm font-medium text-gray-700 mb-1">
                {language === 'es' ? 'Nuevo Estado' : 'New Status'}
              </label>
              <select
                value={selectedStatus}
                onChange={(e) => setSelectedStatus(e.target.value as ProjectStatus)}
                className="w-full rounded-md border-gray-300 shadow-sm"
              >
                <option value="">{language === 'es' ? 'Seleccionar...' : 'Select...'}</option>
                {/* Show ALL statuses for admin override */}
                {['pending_onboarding', 'pending_payment', 'pending_scheduling', 'scheduled',
                  'pending_installation', 'in_progress', 'completed', 'cancelled', 'on_hold',
                  'pending_requirements', 'requirements_defined', 'pending_audit', 'resolved'
                ].map((status) => (
                  <option key={status} value={status}>
                    {getStatusLabel(status as ProjectStatus, language as 'en' | 'es')}
                  </option>
                ))}
              </select>
            </div>

            <div className="mb-4">
              <label className="block text-sm font-medium text-gray-700 mb-1">
                {language === 'es' ? 'Razon del cambio' : 'Reason for change'}
              </label>
              <textarea
                value={overrideReason}
                onChange={(e) => setOverrideReason(e.target.value)}
                rows={3}
                className="w-full rounded-md border-gray-300 shadow-sm"
                placeholder={language === 'es'
                  ? 'Explique por que se necesita este cambio...'
                  : 'Explain why this change is needed...'}
              />
            </div>

            {error && (
              <div className="text-sm text-red-600 mb-4">{error}</div>
            )}

            <div className="flex justify-end gap-3">
              <button
                onClick={() => {
                  setShowOverrideDialog(false);
                  setOverrideReason('');
                  setError(null);
                }}
                className="px-4 py-2 text-gray-600 hover:text-gray-800"
                disabled={isLoading}
              >
                {language === 'es' ? 'Cancelar' : 'Cancel'}
              </button>
              <button
                onClick={() => selectedStatus && handleOverride(selectedStatus)}
                disabled={!selectedStatus || !overrideReason.trim() || isLoading}
                className="px-4 py-2 bg-amber-600 text-white rounded-md hover:bg-amber-700 disabled:bg-gray-300"
              >
                {isLoading
                  ? (language === 'es' ? 'Procesando...' : 'Processing...')
                  : (language === 'es' ? 'Forzar Cambio' : 'Force Change')}
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

export default StatusSelector;
```
